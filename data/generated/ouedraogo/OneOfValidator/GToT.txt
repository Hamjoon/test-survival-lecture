package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.Before;
import org.junit.Test;

import java.util.Set;

import static org.junit.Assert.*;

public class OneOfValidatorTest {

    private ObjectMapper objectMapper;
    private JsonSchemaFactory schemaFactory;

    @Before
    public void setUp() {
        objectMapper = new ObjectMapper();
        schemaFactory = JsonSchemaFactory.getInstance(SpecVersion.VersionFlag.V7);
    }

    @Test
    public void testConstructorAndPreloadJsonSchema() {
        String schemaJson = "{\"oneOf\": [{\"type\": \"string\"}, {\"type\": \"integer\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        assertNotNull(schema);
        // This implicitly tests preloadJsonSchema through initialization
        schema.initializeValidators();
    }

    @Test
    public void testValidateSuccessExactlyOneValidSchema() {
        String schemaJson = "{\"oneOf\": [{\"type\": \"string\"}, {\"type\": \"integer\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        JsonNode node = objectMapper.convertValue("hello", JsonNode.class);
        Set<ValidationMessage> errors = schema.validate(node);
        
        assertTrue("Expected validation to pass with exactly one matching schema", errors.isEmpty());
    }

    @Test
    public void testValidateFailureZeroValidSchemas() {
        String schemaJson = "{\"oneOf\": [{\"type\": \"string\"}, {\"type\": \"integer\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        JsonNode node = objectMapper.convertValue(true, JsonNode.class);
        Set<ValidationMessage> errors = schema.validate(node);
        
        assertFalse("Expected validation to fail when zero schemas match", errors.isEmpty());
    }

    @Test
    public void testValidateFailureMultipleValidSchemas() {
        // A number can match both 'number' and 'integer' if both are in oneOf
        String schemaJson = "{\"oneOf\": [{\"type\": \"integer\"}, {\"type\": \"number\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        JsonNode node = objectMapper.convertValue(5, JsonNode.class);
        Set<ValidationMessage> errors = schema.validate(node);
        
        assertFalse("Expected validation to fail when multiple schemas match", errors.isEmpty());
    }

    @Test(expected = JsonSchemaException.class)
    public void testValidateFailFastException() {
        // Configuring failFast via config or custom setup if applicable, 
        // Here we simulate checking exception handling behavior.
        SchemaValidatorsConfig config = new SchemaValidatorsConfig();
        config.setFailFast(true);
        
        JsonSchemaFactory factoryWithConfig = JsonSchemaFactory.getInstance(SpecVersion.VersionFlag.V7);
        String schemaJson = "{\"oneOf\": [{\"type\": \"integer\"}, {\"type\": \"number\"}]}";
        JsonSchema schema = factoryWithConfig.getSchema(schemaJson, config);
        
        JsonNode node = objectMapper.convertValue(5, JsonNode.class);
        schema.validate(node);
    }

    @Test
    public void testWalkWithValidationEnabled() {
        String schemaJson = "{\"oneOf\": [{\"type\": \"string\"}, {\"type\": \"integer\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        ExecutionContext executionContext = schema.createExecutionContext();
        JsonNode node = objectMapper.convertValue("test", JsonNode.class);
        
        Set<ValidationMessage> errors = schema.walk(executionContext, node, node, "", true);
        assertTrue(errors.isEmpty());
    }

    @Test
    public void testWalkWithValidationDisabled() {
        String schemaJson = "{\"oneOf\": [{\"type\": \"string\"}, {\"type\": \"integer\"}]}";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        ExecutionContext executionContext = schema.createExecutionContext();
        JsonNode node = objectMapper.convertValue(true, JsonNode.class); // invalid for schema, but validation disabled in walk
        
        Set<ValidationMessage> errors = schema.walk(executionContext, node, node, "", false);
        assertTrue("Expected no errors when validation is disabled during walk", errors.isEmpty());
    }
}
