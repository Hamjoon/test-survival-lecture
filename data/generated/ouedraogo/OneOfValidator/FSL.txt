package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.networknt.schema.CollectorContext.Scope;
import org.junit.Before;
import org.junit.Test;

import java.util.Collections;
import java.util.Set;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

public class OneOfValidatorTest {

    private ObjectMapper objectMapper;
    private ValidationContext validationContext;
    private ExecutionContext executionContext;
    private CollectorContext collectorContext;
    private ValidatorState validatorState;

    @Before
    public void setUp() {
        objectMapper = new ObjectMapper();
        validationContext = mock(ValidationContext.class);
        executionContext = mock(ExecutionContext.class);
        collectorContext = mock(CollectorContext.class);
        validatorState = new ValidatorState();

        when(executionContext.getCollectorContext()).thenReturn(collectorContext);
        when(collectorContext.enterDynamicScope()).thenReturn(mock(Scope.class));
        when(collectorContext.get(ValidatorState.VALIDATOR_STATE_KEY)).thenReturn(validatorState);
    }

    @Test
    public void testConstructorAndPreloadJsonSchema() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode());
        JsonSchema mockSubSchema = mock(JsonSchema.class);
        when(validationContext.newSchema(anyString(), any(JsonNode.class), eq(null))).thenReturn(mockSubSchema);

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        validator.preloadJsonSchema();

        verify(mockSubSchema, times(1)).initializeValidators();
    }

    @Test
    public void testValidateExactOneValidSchema() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode()).add(objectMapper.createObjectNode());
        JsonSchema schema1 = mock(JsonSchema.class);
        JsonSchema schema2 = mock(JsonSchema.class);

        when(validationContext.newSchema(eq("#/oneOf/0"), any(JsonNode.class), eq(null))).thenReturn(schema1);
        when(validationContext.newSchema(eq("#/oneOf/1"), any(JsonNode.class), eq(null))).thenReturn(schema2);

        // schema1 passes (empty errors), schema2 fails
        when(schema1.validate(any(), any(), any(), anyString())).thenReturn(Collections.emptySet());
        Set<ValidationMessage> schema2Errors = Collections.singleton(mock(ValidationMessage.class));
        when(schema2.validate(any(), any(), any(), anyString())).thenReturn(schema2Errors);

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.createObjectNode();

        Set<ValidationMessage> errors = validator.validate(executionContext, node, node, "#");

        assertTrue(errors.isEmpty());
        assertTrue(validatorState.isMatchedNode());
        assertFalse(validatorState.isComplexValidator());
    }

    @Test
    public void testValidateMultipleValidSchemas() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode()).add(objectMapper.createObjectNode());
        JsonSchema schema1 = mock(JsonSchema.class);
        JsonSchema schema2 = mock(JsonSchema.class);

        when(validationContext.newSchema(eq("#/oneOf/0"), any(JsonNode.class), eq(null))).thenReturn(schema1);
        when(validationContext.newSchema(eq("#/oneOf/1"), any(JsonNode.class), eq(null))).thenReturn(schema2);

        // Both schemas pass
        when(schema1.validate(any(), any(), any(), anyString())).thenReturn(Collections.emptySet());
        when(schema2.validate(any(), any(), any(), anyString())).thenReturn(Collections.emptySet());

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.createObjectNode();

        Set<ValidationMessage> errors = validator.validate(executionContext, node, node, "#");

        assertFalse(errors.isEmpty());
        assertEquals(1, errors.size());
    }

    @Test
    public void testValidateZeroValidSchemas() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode());
        JsonSchema schema1 = mock(JsonSchema.class);

        when(validationContext.newSchema(eq("#/oneOf/0"), any(JsonNode.class), eq(null))).thenReturn(schema1);

        Set<ValidationMessage> schema1Errors = Collections.singleton(mock(ValidationMessage.class));
        when(schema1.validate(any(), any(), any(), anyString())).thenReturn(schema1Errors);

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.createObjectNode();

        Set<ValidationMessage> errors = validator.validate(executionContext, node, node, "#");

        assertFalse(errors.isEmpty());
        assertTrue(errors.size() >= 2); // OneOf message + child errors
    }

    @Test
    public void testWalkWithShouldValidateSchemaTrue() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode());
        JsonSchema schema1 = mock(JsonSchema.class);

        when(validationContext.newSchema(eq("#/oneOf/0"), any(JsonNode.class), eq(null))).thenReturn(schema1);
        when(schema1.validate(any(), any(), any(), anyString())).thenReturn(Collections.emptySet());

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.createObjectNode();

        Set<ValidationMessage> errors = validator.walk(executionContext, node, node, "#", true);

        assertTrue(errors.isEmpty());
    }

    @Test
    public void testWalkWithShouldValidateSchemaFalse() {
        JsonNode schemaNode = objectMapper.createArrayNode().add(objectMapper.createObjectNode());
        JsonSchema schema1 = mock(JsonSchema.class);

        when(validationContext.newSchema(eq("#/oneOf/0"), any(JsonNode.class), eq(null))).thenReturn(schema1);

        OneOfValidator validator = new OneOfValidator("#/oneOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.createObjectNode();

        Set<ValidationMessage> errors = validator.walk(executionContext, node, node, "#", false);

        assertTrue(errors.isEmpty());
        verify(schema1, times(1)).walk(executionContext, node, node, "#", false);
    }
}
