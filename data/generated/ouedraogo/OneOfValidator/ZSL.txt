package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.Before;
import org.junit.Test;

import java.util.Set;

import static org.junit.Assert.*;

public class OneOfValidatorTest {

    private ObjectMapper objectMapper;
    private JsonSchemaFactory schemaFactory;

    @Before
    public void setUp() {
        objectMapper = new ObjectMapper();
        schemaFactory = JsonSchemaFactory.getInstance(SpecVersion.VersionFlag.V7);
    }

    @Test
    public void testOneOfValidSingleSchema() {
        // One of the schemas should match (e.g., number or string, let's say integer)
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"integer\" }, { \"type\": \"boolean\" } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);

        JsonNode node = objectMapper.createIntNode(5);
        Set<ValidationMessage> errors = schema.validate(node);

        assertTrue("Expected validation to pass when exactly one schema matches", errors.isEmpty());
    }

    @Test
    public void testOneOfInvalidMultipleSchemasMatch() {
        // Both schemas match (e.g., integer and minimum 0)
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"integer\" }, { \"minimum\": 0 } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);

        JsonNode node = objectMapper.createIntNode(5);
        Set<ValidationMessage> errors = schema.validate(node);

        assertFalse("Expected validation to fail when more than one schema matches", errors.isEmpty());
        // Verify that the error corresponds to oneOf failure
        boolean foundOneOfError = errors.stream()
                .anyMatch(msg -> msg.getValidatorType().equals(ValidatorTypeCode.ONE_OF));
        assertTrue("Expected a ONE_OF validation message type", foundOneOfError);
    }

    @Test
    public void testOneOfInvalidZeroSchemasMatch() {
        // None of the schemas match (expecting string or boolean, passing integer)
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"string\" }, { \"type\": \"boolean\" } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);

        JsonNode node = objectMapper.createIntNode(5);
        Set<ValidationMessage> errors = schema.validate(node);

        assertFalse("Expected validation to fail when zero schemas match", errors.isEmpty());
        boolean foundOneOfError = errors.stream()
                .anyMatch(msg -> msg.getValidatorType().equals(ValidatorTypeCode.ONE_OF));
        assertTrue("Expected a ONE_OF validation message type", foundOneOfError);
    }

    @Test
    public void testOneOfWalkWithValidation() {
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"integer\" }, { \"type\": \"boolean\" } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);

        JsonNode node = objectMapper.createIntNode(10);
        ExecutionContext executionContext = schema.createExecutionContext();
        
        // Accessing the validator directly or through walk method with shouldValidateSchema = true
        Set<ValidationMessage> errors = schema.walk(executionContext, node, node, "#", true);
        assertTrue("Expected walk with validation enabled to pass", errors.isEmpty());
    }

    @Test
    public void testOneOfWalkWithoutValidation() {
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"integer\" }, { \"type\": \"boolean\" } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);

        JsonNode node = objectMapper.createIntNode(10);
        ExecutionContext executionContext = schema.createExecutionContext();
        
        // Walk with shouldValidateSchema = false
        Set<ValidationMessage> errors = schema.walk(executionContext, node, node, "#", false);
        assertTrue("Expected walk without validation to return empty messages", errors.isEmpty());
    }

    @Test
    public void testPreloadJsonSchema() {
        String schemaJson = "{ \"oneOf\": [ { \"type\": \"integer\" }, { \"type\": \"string\" } ] }";
        JsonSchema schema = schemaFactory.getSchema(schemaJson);
        
        // Ensure preloadJsonSchema executes without exceptions
        schema.initializeValidators();
        assertTrue(true);
    }
}
