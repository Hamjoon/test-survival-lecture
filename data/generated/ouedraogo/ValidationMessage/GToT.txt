package com.networknt.schema;

import org.junit.Test;
import java.text.MessageFormat;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.*;

public class ValidationMessageTest {

    // Dummy ErrorMessageType implementation for testing static factory methods
    private static class DummyErrorMessageType implements ErrorMessageType {
        private final String errorCode;
        private final String customMessage;

        public DummyErrorMessageType(String errorCode, String customMessage) {
            this.errorCode = errorCode;
            this.customMessage = customMessage;
        }

        @Override
        public String getErrorCode() {
            return errorCode;
        }

        @Override
        public String getCustomMessage() {
            return customMessage;
        }
    }

    @Test
    public void testBasicGettersAndSetters() {
        ValidationMessage msg = new ValidationMessage();
        msg.setType("type1");
        msg.setCode("code1");
        msg.setPath("path1");
        msg.setSchemaPath("schemaPath1");
        msg.setArguments(new String[]{"arg1"});
        Map<String, Object> details = new HashMap<>();
        details.put("key", "value");
        msg.setDetails(details);
        msg.setMessage("message1");

        assertEquals("type1", msg.getType());
        assertEquals("code1", msg.getCode());
        assertEquals("path1", msg.getPath());
        assertEquals("schemaPath1", msg.getSchemaPath());
        assertArrayEquals(new String[]{"arg1"}, msg.getArguments());
        assertEquals(details, msg.getDetails());
        assertEquals("message1", msg.getMessage());
        assertEquals("message1", msg.toString());
    }

    @Test
    public void testBuilderPattern() {
        ValidationMessage msg = new ValidationMessage.Builder()
                .type("error")
                .code("100")
                .path("#/foo")
                .schemaPath("#/properties/foo")
                .arguments("val1", "val2")
                .customMessage("Custom message")
                .build();

        assertEquals("error", msg.getType());
        assertEquals("100", msg.getCode());
        assertEquals("#/foo", msg.getPath());
        assertEquals("#/properties/foo", msg.getSchemaPath());
        assertArrayEquals(new String[]{"val1", "val2"}, msg.getArguments());
        assertEquals("Custom message", msg.getMessage());
    }

    @Test
    public void testBuilderWithFormat() {
        MessageFormat format = new MessageFormat("Path {0} failed with {1}");
        ValidationMessage msg = new ValidationMessage.Builder()
                .path("#/bar")
                .arguments("invalid")
                .format(format)
                .build();

        // format uses path as objs[0] and arguments[0] as objs[1]
        assertEquals("Path #/bar failed with invalid", msg.getMessage());
    }

    @Test
    public void testStaticOfMethods() {
        DummyErrorMessageType errorType = new DummyErrorMessageType("ERR001", "Default custom");
        MessageFormat format = new MessageFormat("Error at {0}");

        ValidationMessage msg1 = ValidationMessage.of("type", errorType, format, "#/at", "#/schema");
        assertNotNull(msg1);
        assertEquals("ERR001", msg1.getCode());
        assertEquals("#/at", msg1.getPath());
        assertEquals("#/schema", msg1.getSchemaPath());
        assertEquals("Error at #/at", msg1.getMessage());

        ValidationMessage msg2 = ValidationMessage.ofWithCustom("type", errorType, format, "Override custom", "#/at", "#/schema", "arg");
        assertEquals("Override custom", msg2.getMessage());

        Map<String, Object> details = new HashMap<>();
        ValidationMessage msg3 = ValidationMessage.of("type", errorType, format, "#/at", "#/schema", details);
        assertEquals(details, msg3.getDetails());
    }

    @Test
    public void testEqualsAndHashCode() {
        ValidationMessage msg1 = new ValidationMessage.Builder()
                .type("type").code("code").path("path").schemaPath("schemaPath").arguments("arg").build();
        ValidationMessage msg2 = new ValidationMessage.Builder()
                .type("type").code("code").path("path").schemaPath("schemaPath").arguments("arg").build();
        ValidationMessage msg3 = new ValidationMessage.Builder()
                .type("different").code("code").path("path").schemaPath("schemaPath").arguments("arg").build();

        assertEquals(msg1, msg1);
        assertEquals(msg1, msg2);
        assertEquals(msg1.hashCode(), msg2.hashCode());
        assertNotEquals(msg1, msg3);
        assertNotEquals(msg1, null);
        assertNotEquals(msg1, "Some String");
    }

    @Test
    public void testEdgeCaseNullArgumentsAndFormat() {
        MessageFormat format = new MessageFormat("Path {0}");
        ValidationMessage msg = new ValidationMessage.Builder()
                .path("#/nullArgs")
                .arguments((String[]) null)
                .format(format)
                .build();

        assertEquals("Path #/nullArgs", msg.getMessage());
    }

    @Test
    public void testEdgeCaseBlankCustomMessage() {
        MessageFormat format = new MessageFormat("Path {0} error");
        ValidationMessage msg = new ValidationMessage.Builder()
                .path("#/test")
                .customMessage("   ") // blank message should fallback to format
                .format(format)
                .build();

        assertEquals("Path #/test error", msg.getMessage());
    }
}
