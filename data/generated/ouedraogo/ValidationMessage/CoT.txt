package com.networknt.schema;

import org.junit.Test;
import java.text.MessageFormat;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.*;

public class ValidationMessageTest {

    @Test
    public void testGettersAndSetters() {
        ValidationMessage msg = new ValidationMessage();
        msg.setType("type");
        msg.setCode("100");
        msg.setPath("#/path");
        msg.setSchemaPath("#/schemaPath");
        String[] args = new String[]{"arg1"};
        msg.setArguments(args);
        Map<String, Object> details = new HashMap<>();
        details.put("key", "value");
        msg.setDetails(details);
        msg.setMessage("test message");

        assertEquals("type", msg.getType());
        assertEquals("100", msg.getCode());
        assertEquals("#/path", msg.getPath());
        assertEquals("#/schemaPath", msg.getSchemaPath());
        assertArrayEquals(args, msg.getArguments());
        assertEquals(details, msg.getDetails());
        assertEquals("test message", msg.getMessage());
    }

    @Test
    public void testToString() {
        ValidationMessage msg = new ValidationMessage();
        msg.setMessage("Hello World");
        assertEquals("Hello World", msg.toString());
    }

    @Test
    public void testEqualsAndHashCode() {
        ValidationMessage msg1 = new ValidationMessage.Builder()
                .type("type")
                .code("100")
                .path("#/path")
                .schemaPath("#/schemaPath")
                .arguments("arg1")
                .build();

        ValidationMessage msg2 = new ValidationMessage.Builder()
                .type("type")
                .code("100")
                .path("#/path")
                .schemaPath("#/schemaPath")
                .arguments("arg1")
                .build();

        ValidationMessage msg3 = new ValidationMessage.Builder()
                .type("different")
                .code("100")
                .path("#/path")
                .schemaPath("#/schemaPath")
                .arguments("arg1")
                .build();

        assertEquals(msg1, msg1); // reflexivity
        assertEquals(msg1, msg2);
        assertEquals(msg1.hashCode(), msg2.hashCode());
        assertNotEquals(msg1, msg3);
        assertNotEquals(msg1, null);
        assertNotEquals(msg1, "some string");
    }

    @Test
    public void testStaticOfMethods() {
        DummyErrorMessageType errorType = new DummyErrorMessageType("100", "Custom Message");
        MessageFormat format = new MessageFormat("Path {0} failed with {1}");

        ValidationMessage msg1 = ValidationMessage.of("type", errorType, format, "#/path", "#/schemaPath", "arg1");
        assertNotNull(msg1);
        assertEquals("100", msg1.getCode());
        assertEquals("#/path", msg1.getPath());
        assertEquals("#/schemaPath", msg1.getSchemaPath());
        assertArrayEquals(new String[]{"arg1"}, msg1.getArguments());
        assertEquals("Custom Message", msg1.getMessage());

        ValidationMessage msg2 = ValidationMessage.ofWithCustom("type", errorType, format, "Override Message", "#/path", "#/schemaPath", "arg1");
        assertNotNull(msg2);
        assertEquals("Override Message", msg2.getMessage());

        Map<String, Object> details = new HashMap<>();
        ValidationMessage msg3 = ValidationMessage.of("type", errorType, format, "#/path", "#/schemaPath", details);
        assertNotNull(msg3);
        assertEquals(details, msg3.getDetails());
    }

    @Test
    public void testEqualsAndHashCodeEdgeCases() {
        ValidationMessage msg1 = new ValidationMessage();
        ValidationMessage msg2 = new ValidationMessage();

        // Both all null fields
        assertEquals(msg1, msg2);
        assertEquals(msg1.hashCode(), msg2.hashCode());

        // One field differs (type)
        msg1.setType("A");
        msg2.setType("B");
        assertNotEquals(msg1, msg2);

        // Code differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        msg1.setCode("1");
        msg2.setCode("2");
        assertNotEquals(msg1, msg2);

        // Path differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        msg1.setPath("1");
        msg2.setPath("2");
        assertNotEquals(msg1, msg2);

        // SchemaPath differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        msg1.setSchemaPath("1");
        msg2.setSchemaPath("2");
        assertNotEquals(msg1, msg2);

        // Details differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        Map<String, Object> map1 = new HashMap<>();
        map1.put("k", "v");
        msg1.setDetails(map1);
        assertNotEquals(msg1, msg2);

        // Arguments differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        msg1.setArguments(new String[]{"a"});
        msg2.setArguments(new String[]{"b"});
        assertNotEquals(msg1, msg2);

        // Message differs
        msg1 = new ValidationMessage();
        msg2 = new ValidationMessage();
        msg1.setMessage("a");
        msg2.setMessage("b");
        assertNotEquals(msg1, msg2);
    }

    @Test
    public void testBuilderEdgeCases() {
        // Empty builder
        ValidationMessage msg = new ValidationMessage.Builder().build();
        assertNotNull(msg);
        assertNull(msg.getType());
        assertNull(msg.getCode());
        assertNull(msg.getPath());
        assertNull(msg.getSchemaPath());
        assertNull(msg.getArguments());
        assertNull(msg.getDetails());
        assertNull(msg.getMessage());

        // Format with null arguments
        MessageFormat format = new MessageFormat("Test {0}");
        ValidationMessage msgFormatNullArgs = new ValidationMessage.Builder()
                .path("#/path")
                .format(format)
                .arguments((String[]) null)
                .build();
        assertEquals("Test #/path", msgFormatNullArgs.getMessage());
    }

    // Helper implementation for ErrorMessageType testing
    private static class DummyErrorMessageType implements ErrorMessageType {
        private final String errorCode;
        private final String customMessage;

        public DummyErrorMessageType(String errorCode, String customMessage) {
            this.errorCode = errorCode;
            this.customMessage = customMessage;
        }

        @Override
        public String getErrorCode() {
            return errorCode;
        }

        @Override
        public String getCustomMessage() {
            return customMessage;
        }
    }
}
