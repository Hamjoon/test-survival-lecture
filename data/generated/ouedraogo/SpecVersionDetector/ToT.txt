package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.networknt.schema.SpecVersion.VersionFlag;
import org.junit.Test;

import java.nio.file.Paths;
import java.util.Optional;

import static org.junit.Assert.*;

public class SpecVersionDetectorTest {

    private final ObjectMapper objectMapper = new ObjectMapper();

    @Test
    public void testDetectValidSchema() throws Exception {
        JsonNode node = objectMapper.readTree("{\"$schema\": \"https://json-schema.org/draft/2020-12/schema\"}");
        VersionFlag version = SpecVersionDetector.detect(node);
        assertEquals(VersionFlag.V202012, version);
    }

    @Test(expected = JsonSchemaException.class)
    public void testDetectMissingSchemaTag() throws Exception {
        JsonNode node = objectMapper.readTree("{}");
        SpecVersionDetector.detect(node);
    }

    @Test
    public void testDetectOptionalVersionPresent() throws Exception {
        JsonNode node = objectMapper.readTree("{\"$schema\": \"http://json-schema.org/draft-07/schema#\"}");
        Optional<VersionFlag> version = SpecVersionDetector.detectOptionalVersion(node);
        assertTrue(version.isPresent());
        assertEquals(VersionFlag.V7, version.get());
    }

    @Test
    public void testDetectOptionalVersionAbsent() throws Exception {
        JsonNode node = objectMapper.readTree("{}");
        Optional<VersionFlag> version = SpecVersionDetector.detectOptionalVersion(node);
        assertFalse(version.isPresent());
    }

    @Test(expected = JsonSchemaException.class)
    public void testDetectOptionalVersionUnrecognizable() throws Exception {
        JsonNode node = objectMapper.readTree("{\"$schema\": \"http://unknown-schema.org/draft/schema\"}");
        SpecVersionDetector.detectOptionalVersion(node);
    }

    @Test
    public void testDetectVersionFromNodePathAndDefault() throws Exception {
        // Node has no schema, but path contains supported version, and default is provided
        JsonNode node = objectMapper.readTree("{}");
        Optional<VersionFlag> versionFromPath = SpecVersionDetector.detectVersionFromPath(Paths.get("/some/path/draft4/test"));
        assertTrue(versionFromPath.isPresent());
        assertEquals(VersionFlag.V4, versionFromPath.get());

        VersionFlag version = SpecVersionDetector.detectVersion(node, Paths.get("/some/path/draft6/test"), VersionFlag.V201909);
        assertEquals(VersionFlag.V6, version);

        // Fallback to default when neither node nor path provides it
        VersionFlag defaultVersion = SpecVersionDetector.detectVersion(node, Paths.get("/some/path/unknown/test"), VersionFlag.V201909);
        assertEquals(VersionFlag.V201909, defaultVersion);
    }

    @Test
    public void testDetectOptionalVersionString() {
        Optional<VersionFlag> version = SpecVersionDetector.detectOptionalVersion("https://json-schema.org/draft/2019-09/schema");
        assertTrue(version.isPresent());
        assertEquals(VersionFlag.V201909, version.get());

        Optional<VersionFlag> unknown = SpecVersionDetector.detectOptionalVersion("http://example.com/unknown");
        assertFalse(unknown.isPresent());
    }
}
