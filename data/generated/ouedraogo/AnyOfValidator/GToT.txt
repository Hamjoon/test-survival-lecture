package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.Before;
import org.junit.Test;

import java.util.Collections;
import java.util.Set;

import static org.junit.Assert.*;

/**
 * JUnit 4 test cases for AnyOfValidator covering basic functionality, edge cases, 
 * OpenAPI discriminators, and exception scenarios.
 */
public class AnyOfValidatorTest {

    private ObjectMapper objectMapper;
    private SchemaValidatorsConfig config;
    private ValidationContext validationContext;

    @Before
    public void setUp() {
        objectMapper = new ObjectMapper();
        config = new SchemaValidatorsConfig();
        validationContext = new ValidationContext(config, (schemaUri, reference) -> null);
    }

    @Test
    public void testConstructorAndBasicFunctionality() throws Exception {
        // Test constructor and basic validation path
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}, {\"type\": \"integer\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);
        
        assertNotNull(validator);
        validator.preloadJsonSchema();
    }

    @Test
    public void testValidateBasicMatch() throws Exception {
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}, {\"type\": \"integer\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);

        ExecutionContext executionContext = validationContext.newExecutionContext();
        JsonNode node = objectMapper.readTree("\"testString\"");
        
        // Setup mock ValidatorState in CollectorContext
        ValidatorState state = new ValidatorState();
        executionContext.getCollectorContext().set(ValidatorState.VALIDATOR_STATE_KEY, state);

        Set<ValidationMessage> errors = validator.validate(executionContext, node, node, "#");
        assertTrue("Expected no errors when matching one of the schemas", errors.isEmpty());
    }

    @Test
    public void testWalkMethodWithoutValidation() throws Exception {
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);

        ExecutionContext executionContext = validationContext.newExecutionContext();
        JsonNode node = objectMapper.readTree("\"testString\"");
        
        ValidatorState state = new ValidatorState();
        executionContext.getCollectorContext().set(ValidatorState.VALIDATOR_STATE_KEY, state);

        Set<ValidationMessage> errors = validator.walk(executionContext, node, node, "#", false);
        assertNotNull(errors);
        assertTrue(errors.isEmpty());
    }

    @Test
    public void testWalkMethodWithValidation() throws Exception {
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);

        ExecutionContext executionContext = validationContext.newExecutionContext();
        JsonNode node = objectMapper.readTree("\"testString\"");
        
        ValidatorState state = new ValidatorState();
        executionContext.getCollectorContext().set(ValidatorState.VALIDATOR_STATE_KEY, state);

        Set<ValidationMessage> errors = validator.walk(executionContext, node, node, "#", true);
        assertNotNull(errors);
        assertTrue(errors.isEmpty());
    }

    @Test(expected = NullPointerException.class)
    public void testEdgeCaseNullExecutionContext() throws Exception {
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);
        JsonNode node = objectMapper.readTree("\"testString\"");

        // Passing null execution context to trigger NullPointerException inside validate
        validator.validate(null, node, node, "#");
    }

    @Test
    public void testEdgeCaseNoMatchFound() throws Exception {
        JsonNode schemaNode = objectMapper.readTree("[{\"type\": \"string\"}, {\"type\": \"integer\"}]");
        AnyOfValidator validator = new AnyOfValidator("#/anyOf", schemaNode, null, validationContext);

        ExecutionContext executionContext = validationContext.newExecutionContext();
        JsonNode node = objectMapper.readTree("true"); // boolean matches neither string nor integer
        
        ValidatorState state = new ValidatorState();
        executionContext.getCollectorContext().set(ValidatorState.VALIDATOR_STATE_KEY, state);

        Set<ValidationMessage> errors = validator.validate(executionContext, node, node, "#");
        assertFalse("Expected validation errors when no subschema matches", errors.isEmpty());
    }
}
