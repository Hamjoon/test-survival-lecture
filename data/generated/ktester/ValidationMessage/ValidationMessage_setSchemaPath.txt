    }

        String schemaPath = "#/properties/field/schemaPath";

        validationMessage.setSchemaPath(schemaPath);
        // then
        assertEquals(schemaPath, validationMessage.getSchemaPath());
    }

        // then
        assertEquals(schemaPath, validationMessage.getSchemaPath(), "Verify that schemaPath is correctly set to the expected string");
    }

    @Test
    void testSetSchemaPath_Null() {
        // given
        String schemaPath = null;
        // when
        validationMessage.setSchemaPath(schemaPath);
        // then
        assertNull(validationMessage.getSchemaPath(), "Verify that schemaPath is correctly set to null");
    }

    @Test
    void testSetSchemaPath_EmptyString() {
        // given

        // when
